## โจทย์ข้อที่ 53: ระบบบริหารคลาสและวิเคราะห์รายได้ฟิตเนสคลับ (Smart Fitness Club Analytics Engine)

## input {
  interface FitnessSession {
  sessionId: string;                                    // รหัสคลาส (เช่น "FIT-01", "FIT-02")
  trainerName: string;                                  // ชื่อเทรนเนอร์ผู้สอน (เช่น "Coach Alex", "Coach Sarah", "Coach John")
  classType: "YOGA" | "SPINNING" | "CROSSFIT";          // ประเภทคลาส (250 | 400 | 600 บาท ต่อผู้เข้าเรียน 1 คน)
  attendeeCount: number;                                // จำนวนผู้เข้าเรียนในคลาสนั้น (คน)
  includesNutritionPlan?: boolean;                      // คลาสนี้มีแพ็กเกจอาหารคลีน/เวย์โปรตีนเสริมหรือไม่ (ถ้า true +300 บาทต่อคลาส!)
  isTrainerAbsent?: boolean;                            // เทรนเนอร์ลาป่วย/คลาสถูกยกเลิกหรือไม่ (ถ้า true ให้ข้าม ไม่คิดเงิน!)
}

## }

## output {
  totalClubRevenue: number;         // รายได้รวมสุทธิทั้งหมดของคลับ (ค่าคลาสตามจำนวนคน + ค่าแพ็กเกจอาหารเสริม)
  totalCompletedClasses: number;    // จำนวนคลาสที่เปิดสอนสำเร็จ
  cancelledClassesCount: number;    // จำนวนคลาสที่ถูกยกเลิก
  totalAttendees: number;           // จำนวนผู้เข้าเรียนรวมทั้งหมดในคลาสที่สำเร็จ (คน)
  nutritionPlanRevenue: number;     // รายได้เฉพาะค่าแพ็กเกจอาหารคลีน/โปรตีนเสริมทั้งหมด
  topGrossingTrainer: string;       // ชื่อเทรนเนอร์ที่สร้างรายได้รวมให้คลับสูงสุดอันดับ 1
## }

## Process {
  1. ถ้าข้อมูลที่รับเข้ามา = 0 คืนค่าตั้งต้นทันที

  2.  ประกาศ cancelledClassesCount สะสมยอดไม่สำเร็จ
      ประกาศ  totalCompletedClasses สะสมยอดสำเร็จ
  -> คลาสถูกยกเลิกหรือไม่ ? (isTrainerAbsent === true?)
  yes -> ข้าม และสะสมยอดถูกยกเลิก cancelledClassesCount += 1
  no -> นำมาคำณวนต่อและสะสมยอดสำเร็จ totalCompletedClasses += 1
  
  3. ประกาศ  classRate เก็บเงินคลาสนั้นชั่วคราวเพื่อใช้หาผลรวม
    ประกาศ classCost เก็บผลรวมต่อคลาสชั่วคราวเพื่อใช้หาผลรวม
  -> เข้าคลาสอะไร? (classType?)
  -> "YOGA" ➔ 250 บาท ต่อคน / classRate = 250
  -> "SPINNING" ➔ 400 บาท ต่อคน / classRate = 400
  -> "CROSSFIT" ➔ 600 บาท ต่อคน / classRate = 600
  -> classCost = classRate * attendeeCount

  4. ประกาศ nutritionPlanRevenue สะสมยอดเงินค่าแพ็กเกจอาหารคลีน
  -> รับอาหารคลีนด้วยหรือไม่? (includesNutritionPlan?)
  yes -> nutritionPlanRevenue += 300 AND classCost + 300
  no -> nutritionPlanRevenue += 0 AND classCost + 0

  5. ประกาศ totalClubRevenue เก็บรายได้ทั้งหมด
    ประกาศ totalAttendees เก็บผลรวมคนสมัครคลาสทั้งหมด
    ประกาศ trainerRevenueMap[trainerName] เพื่อเก็บรายได้รวมแยกตามชื่อโค้ช
    totalClubRevenue += classCost
    totalAttendees += attendeeCount

    เทรนเนอร์คนนี้มีชื่อในออฟเจ็คยัง? !trainerRevenueMap[trainerName]?
    yes -> trainerRevenueMap[trainerName] = 0
    no -> trainerRevenueMap[trainerName] += classCost

  6. ประกาศ trainerName เพื่อเก็บชื่อโค้ชทั้งหมดจาก trainerRevenueMap[trainerName]
    ประกาศ  topGrossingTrainer เก็บชื่อเทรนเนอร์ที่ทำเงินที่สุด
    ประกาศ maxCost เก็บยอดเงินชองโค้ชที่ทำเงินสูงสุด
    ยอดเงินของแชมป์ < ยอดเงินของคนนี้หรือเปล่า
    yes -> ยอดเงินของคนนี้ = ยอดเงินของแชมป์ (maxCost = trainerRevenueMap[trainerName])
           ชื่อของแชมป์ = ชื่อของโค้ชคนนี้ (topGrossingTrainer = trainerName)
    no -> ข้าม 
  
  7. คืนค่า 
    {
  totalClubRevenue,
  totalCompletedClasses,
  cancelledClassesCount,
  totalAttendees,
  nutritionPlanRevenue,
  topGrossingTrainer,
}

  
## }

## Variable {
1. ตัวแปรหลัก ที่ต้องคืนค่า
  totalClubRevenue
  totalCompletedClasses
  cancelledClassesCount
  totalAttendees
  nutritionPlanRevenue
  topGrossingTrainer
  trainerRevenueMap[trainerName]

2. ตัวแปรชั่วคราว
  classRate
  classCost
  trainerName
  maxCost
## }
