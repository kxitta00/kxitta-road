const rawData: string = " Somchai:85 , Somsri:-5 , Sombat:40 , Unknown:abc , Somying:102 , Somsak:75 , Nate:0 ";


input {
  สิ่งที่ต้องรับเข้า  = รับชื่อนักเรียน, คะแนนนักเรียนที่เป็นจำนวนเต็มไม่น้อยกว่า 0 และไม่มากกว่า 100
  
  !! สิ่งที่ต้องกำจัดและระวัง  = ช่องว่าง, ตัวเลขติดลบ, ตัวเลขที่มากกว่า 100, กำจัด ":" จากชุดข้อความ, ระวัง!! ชื่อที่ไม่ใช่ชื่อคน ระวังเลขที่ไม่ใช่ตัวเลข;
}

processing Pipeline {
  1.รับข้อมูลมาคลีน = {
    1.ทำเป็นอาเรย์ตัดด้วย "," 
    2.แต่ละแถวตัดด้วย ":" แล้ว trim ช่องว่างทั้งชื่อและคะแนน
    3.คิดทิ้งเฉพาะแถวทีพัง
      แปลงคะแนนด้วย Number แล้วเข็คว่า !isNaN
      และคะแนนต้องอยู่ระหว่าง 0 -> 100
    4.นำแถวที่ผ่านเข้าสู่โครงสร้าง Student
    5.นำไปตัดสินคะแนนแล้วสรุปผล
  }
  2.นำข้อมูลมาตัดเกรดและแสดงผลสรุป
}

Output Shape {
  นำเข้าข้อมูลจากฟังก์ชันแปลงข้อมูลดิบด้วย type Student {
    name,
    score,
  }
  ส่งออกข้อมูลด้วย type report{
    totalProcessed,
    gradeDistribution = ABCDF
    passedCount จำรวนคนที่ได้เกร ABCD
    failCount จำนวนคนที่ได้ F
  }
}
